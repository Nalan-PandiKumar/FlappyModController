struct Player
{
    int y;   // alignment 4 
    int velocity; // alignment 4
};

default alignment -> 8

offsets:-
0-3 -> y
4->7 -> velocity
size -> 8 bytes => 8 % largest alignment of player is 4 => 0 

struct Pipe
{

    int gapY;  // alignment 4 
    int gapSize;  // alignment 4 
    int width;   // alignment 4

};

offsets:-
0-3 -> x
4-7 -> gapY
8-11 -> gapSize
12-15 -> width
16 -> passed
17-19 -> tail padding
size -> 20 bytes => 20 % largest alignment of player is 4 => 0
